Sub checktabulation() '检查填写表
startrow1 = Sheet13.[b1048576].End(3).Row
For i = 2 To startrow1
    printing = Sheet13.Range("A" & i)
    strcode = Sheet13.Range("B" & i)
    colorcode = Sheet13.Range("C" & i)
    colorarr = Split(Sheet13.Range("C" & i), "/")
    Size = Sheet13.Range("D" & i)
    Sizearr = Split(Sheet13.Range("D" & i), "/")
    '印花检查
    startrow2 = Sheet9.[b1048576].End(3).Row
    If printing = "" Then
        streror = "印花不可填写为空，如款式无印花，请填写‘TM’/Printing cannot be left blank. If the style has no printing, please fill in 'TM'"
        MsgBox (streror)
        Sheet13.Range("A" & i).Activate
        Exit Sub
    Else
        printmup = Split(printing, "+")
        For k = 0 To UBound(printmup)
            For l = 2 To startrow2
                If printmup(k) = Sheet9.Range("B" & l) Then
                    GoTo skip1
                End If
            Next l
        streror = printing & " 的组成印花 " & printmup(k) & " 在印花库中不存在/ " & printing & " 's Composition Printing " & printmup(k) & " Does not exist in the Print library"
        MsgBox (streror)
        Sheet13.Range("A" & i).Activate
        Exit Sub
skip1:
        Next k
    End If
    '款式检查
    startrow3 = Sheet10.[a1048576].End(3).Row
    If strcode = "" Then
        streror = "款式不可填写为空/Clothing cannot be left blank."
        MsgBox (streror)
        Sheet13.Range("B" & i).Activate
        Exit Sub
    Else
        strmup = Split(strcode, "+")
        For k = 0 To UBound(strmup)
            For l = 2 To startrow3
                If strmup(k) = Sheet10.Range("B" & l) Then
                    GoTo skip3
                End If
            Next l
        streror = strcode & "的组成款式 " & strmup(k) & " 在款式库中不存在/ " & strcode & " 's Composition Clothing " & strmup(k) & " Does not exist in the Clothing library"
        MsgBox (streror)
        Sheet13.Range("B" & i).Activate
        Exit Sub
skip3:
        Next k
    End If
    '颜色检查及组合项个数检查
    startrow4 = Sheet8.[a1048576].End(3).Row
    If colorcode = "" Then
        streror = "颜色不可填写为空/Color cannot be left blank."
        MsgBox (streror)
        Sheet13.Range("C" & i).Activate
        Exit Sub
    Else
        For Z = 0 To UBound(colorarr)
            colormup = Split(colorarr(Z), "+")
            For k = 0 To UBound(colormup)
                For l = 2 To startrow4
                    If colormup(k) = Sheet8.Range("B" & l) Then
                        GoTo Skip5
                    End If
                Next l
            streror = colorarr(Z) & " 的组成颜色 " & colormup(k) & " 在颜色库中不存在/ " & colorarr(Z) & " 's Composition Color " & colormup(k) & " Does not exist in the Color library"
            MsgBox (streror)
            Sheet13.Range("C" & i).Activate
            Exit Sub
Skip5:
            Next k
            If Not UBound(printmup) = UBound(strmup) Or Not UBound(printmup) = UBound(colormup) Or Not UBound(strmup) = UBound(colormup) Then
                streror = printing & "," & strcode & "," & colorarr(Z) & "的组合项个数不一致/" & printing & "," & strcode & "," & colorarr(Z) & "'s number of combination items is inconsistent"
                MsgBox (streror)
                Sheet13.Range("B" & i).Activate
                Exit Sub
            End If
        Next Z
    End If
    '尺码检查
    If Size = "" Then
        streror = "尺码不可填写为空/Size cannot be left blank."
        MsgBox (streror)
        Sheet13.Range("D" & i).Activate
        Exit Sub
    End If
    '颜色精细化检查
    For k = 0 To UBound(strmup)
        For l = 2 To startrow3
            If strmup(k) = Sheet10.Range("B" & l) Then
                clo = Sheet10.Range("C" & l)
                colarr = Split(clo, "/")
                For Z = 0 To UBound(colorarr)
                    colormup = Split(colorarr(Z), "+")
                    For x = 0 To UBound(colarr)
                        If colormup(k) = colarr(x) Then
                            GoTo Skip7
                        End If
                    Next x
                    streror = "颜色" & colormup(k) & "在款式" & strmup(k) & "中不存在/Color " & colormup(k) & " In Clothing " & strmup(k) & " Does not exist"
                    MsgBox (streror)
                    Sheet13.Range("C" & i).Activate
                    Exit Sub
Skip7:
                Next Z
            End If
            Next l
        Next k
    '尺码精细化检查
    For k = 0 To UBound(strmup)
        For l = 2 To startrow3
            If strmup(k) = Sheet10.Range("B" & l) Then
            For Z = 0 To UBound(Sizearr)
                siz = Sheet10.Range("E" & l)
                sizarr = Split(siz, "/")
                For x = 0 To UBound(sizarr)
                    If Sizearr(Z) = sizarr(x) Then
                        GoTo Skip9
                    End If
                Next x
                streror = "尺码 " & Sizearr(Z) & " 在款式 " & strmup(k) & " 中不存在/Size " & Sizearr(Z) & " In Clothing " & strmup(k) & " Does not exist"
                MsgBox (streror)
                Sheet13.Range("D" & i).Activate
                Exit Sub
Skip9:
            Next Z
            End If
        Next l
    Next k
    
Next i
streror = "检查无误/Check Right"
MsgBox (streror)
End Sub

Sub creatsku() '生成商品编码
startrowR = 2
startrow1 = Sheet13.[e1048576].End(3).Row
If startrow1 > 1 Then Sheet13.Range("e2:e" & startrow1).ClearContents
startrow2 = Sheet13.[b1048576].End(3).Row
For i = 2 To startrow2

printing = Sheet13.Range("A" & i)
printmup = Split(printing, "+")
printcode = ""
startrowfors = Sheet9.[b1048576].End(3).Row
If UBound(printmup) = 0 Then
    printcode = printing
Else
    For Z = 0 To UBound(printmup)
        For a = 2 To startrowfors
            If printmup(Z) = Sheet9.Range("B" & a) Then
                printcode = printcode & Sheet9.Range("A" & a)
            End If
        Next a
    Next Z
End If

strcode = Sheet13.Range("B" & i)
strmup = Split(strcode, "+")
strcomcode = ""
startrowfors = Sheet10.[b1048576].End(3).Row
If UBound(strmup) = 0 Then
    strcomcode = strcode
Else
    For Z = 0 To UBound(strmup)
        For a = 2 To startrowfors
            If strmup(Z) = Sheet10.Range("B" & a) Then
                strcomcode = strcomcode & Sheet10.Range("A" & a)
            End If
        Next a
    Next Z
End If

colorarr = Split(Sheet13.Range("C" & i), "/")
Sizearr = Split(Sheet13.Range("D" & i), "/")
startrow3 = Sheet8.[a1048576].End(3).Row
For l = 0 To UBound(colorarr)
    colormup = Split(colorarr(l), "+")
    colorcode = ""
    If UBound(colormup) = 0 Then
        colorcode = colorarr(l)
    Else
        For Z = 0 To UBound(colormup)
            For j = 2 To startrow3
                If colormup(Z) = Sheet8.Range("B" & j) Then
                    colorcode = colorcode & Sheet8.Range("A" & j)
                End If
            Next j
        Next Z
    End If
    For m = 0 To UBound(Sizearr)
        code = printcode & "-" & strcomcode & "-" & colorcode & "-" & Sizearr(m)
        Sheet13.Range("E" & startrowR) = code
        startrowR = startrowR + 1
    Next m
Next l
Next i
End Sub

        For Z = 0 To UBound(colormup)
            For j = 2 To startrow3
                If colormup(Z) = Sheet8.Range("B" & j) Then
                    colorcode = colorcode & Sheet8.Range("A" & j)
                End If
            Next j
        Next Z
    End If
    For m = 0 To UBound(Sizearr)
        startrowRl = 9
        code = printcode & "-" & strcomcode & "-" & colorcode & "-" & Sizearr(m)
        stcode = ""
        For y = 0 To UBound(strmup)
            If y = 0 Then
                stcode = printmup(y) & "-" & strmup(y) & "-" & colormup(y) & "-" & Sizearr(m)
                stcodename = dicyh(printmup(y)) & "/" & dicks(strmup(y)) & "/" & dicys(strmup(y) & colormup(y)) & "/" & Sizearr(m)
            Else
                stcode = stcode & "/" & printmup(y) & "-" & strmup(y) & "-" & colormup(y) & "-" & Sizearr(m)
                stcodename = stcodename & "&" & dicyh(printmup(y)) & "/" & dicks(strmup(y)) & "/" & dicys(strmup(y) & colormup(y)) & "/" & Sizearr(m)
            End If
        Next y
        'TM为纯色
        If Not UBound(strmup) = 0 Or Not printing = "TM" Then
            For y = 0 To UBound(strmup)
                Sheet12.Range("A" & startrowR) = code
                Sheet12.Range("B" & startrowR) = stcodename
                '修改仓库
                Sheet12.Range("C" & startrowR) = "泰国仓"
                '修改组合sku分类
                Sheet12.Range("D" & startrowR) = "泰国SKU"
                'TM- 拼接sku
                Sheet12.Cells(startrowR, startrowRl) = "TM-" & strmup(y) & "-" & colormup(y) & "-" & Sizearr(m)
                Sheet12.Cells(startrowR, startrowRl + 1) = "1"
                startrowRl = startrowRl + 3
                'TM为纯色
                If Not printing = "TM" Then
                'TM为纯色
                    If Not printmup(y) = "TM" Then
                        Sheet12.Cells(startrowR, startrowRl) = printmup(y)
                        Sheet12.Cells(startrowR, startrowRl + 1) = "1"
                        startrowRl = startrowRl + 3
                    End If
                End If
            Next y
        startrowR = startrowR + 1
        End If
    Next m
Next l
Next i
Sheet12.Activate
Sheet12.UsedRange.RemoveDuplicates Columns:=1, Header:=xlYes
Sheet12.Range("A1").Select
